3) src/lib/soul-codex/people/completeness.ts

import type { PersonDataCompleteness, PersonRegistryEntry } from "./types";

function hasNonEmpty(value?: string): boolean {
  return typeof value === "string" && value.trim().length > 0;
}

export function getPersonDataCompleteness(
  person: PersonRegistryEntry
): PersonDataCompleteness {
  const hasBirthDate = hasNonEmpty(person.birth_date);
  const hasBirthTime = hasNonEmpty(person.birth_time);
  const hasBirthLocation = hasNonEmpty(person.birth_location);

  return {
    hasBirthDate,
    hasBirthTime,
    hasBirthLocation,
    canRunFullNatal: hasBirthDate && hasBirthTime && hasBirthLocation,
    canRunRisingAndHouses: hasBirthDate && hasBirthTime && hasBirthLocation,
    canRunHumanDesign: hasBirthDate && hasBirthTime && hasBirthLocation,
    canRunNumerology: hasBirthDate,
  };
}

4) src/lib/soul-codex/people/confidence.ts

import type { PersonRegistryEntry, RegistryConfidence } from "./types";
import { getPersonDataCompleteness } from "./completeness";

export function deriveRegistryConfidence(
  person: Omit<PersonRegistryEntry, "confidence">
): RegistryConfidence {
  const completeness = getPersonDataCompleteness({
    ...person,
    confidence: "unverified",
  });

  if (
    completeness.hasBirthDate &&
    completeness.hasBirthTime &&
    completeness.hasBirthLocation
  ) {
    return "partial";
  }

  if (completeness.hasBirthDate) {
    return "unverified";
  }

  return "unverified";
}

export function confidenceBadgeLabel(confidence: RegistryConfidence): string {
  switch (confidence) {
    case "verified":
      return "Verified";
    case "partial":
      return "Partial";
    case "unverified":
      return "Unverified";
    default:
      return "Unknown";
  }
}

5) src/lib/soul-codex/people/fallbacks.ts

This is the important part. It tells the app what it is allowed to show when birth data is incomplete, instead of hallucinating a rising sign like a drunk carnival psychic.

import type { PersonDataCompleteness } from "./types";

export interface AllowedSystems {
  showSunSign: boolean;
  showMoonSign: boolean;
  showRising: boolean;
  showHouses: boolean;
  showHumanDesign: boolean;
  showNumerology: boolean;
  showFullProfile: boolean;
  showPartialProfile: boolean;
}

export function getAllowedSystems(
  completeness: PersonDataCompleteness
): AllowedSystems {
  const full = completeness.canRunFullNatal;

  return {
    showSunSign: completeness.hasBirthDate,
    showMoonSign: full,
    showRising: completeness.canRunRisingAndHouses,
    showHouses: completeness.canRunRisingAndHouses,
    showHumanDesign: completeness.canRunHumanDesign,
    showNumerology: completeness.canRunNumerology,
    showFullProfile: full,
    showPartialProfile: completeness.hasBirthDate,
  };
}

6) src/lib/soul-codex/people/profileMode.ts

import type { PersonRegistryEntry } from "./types";
import { getPersonDataCompleteness } from "./completeness";
import { getAllowedSystems } from "./fallbacks";

export type ProfileMode = "full" | "partial" | "placeholder";

export function resolveProfileMode(person: PersonRegistryEntry): ProfileMode {
  const completeness = getPersonDataCompleteness(person);
  const allowed = getAllowedSystems(completeness);

  if (allowed.showFullProfile) return "full";
  if (allowed.showPartialProfile) return "partial";
  return "placeholder";
}

8) src/lib/soul-codex/people/placeholderBuilder.ts

This gives you a clean shell when data is incomplete.

import type { PersonRegistryEntry } from "./types";
import { getPersonDataCompleteness } from "./completeness";
import { getAllowedSystems } from "./fallbacks";
import { resolveProfileMode } from "./profileMode";

export function buildPersonPlaceholder(person: PersonRegistryEntry) {
  const completeness = getPersonDataCompleteness(person);
  const allowed = getAllowedSystems(completeness);
  const mode = resolveProfileMode(person);

  return {
    id: person.id,
    label: person.label,
    relationship: person.relationship,
    confidence: person.confidence,
    mode,
    input_status: {
      birth_date: completeness.hasBirthDate,
      birth_time: completeness.hasBirthTime,
      birth_location: completeness.hasBirthLocation,
    },
    allowed_systems: allowed,
    notes: person.notes || null,
    missing_fields: [
      !completeness.hasBirthDate ? "birth_date" : null,
      !completeness.hasBirthTime ? "birth_time" : null,
      !completeness.hasBirthLocation ? "birth_location" : null,
    ].filter(Boolean),
  };
}

11) src/components/soul-codex/PeopleRegistryCard.tsx

type PersonCard = {
  id: string;
  label: string;
  relationship?: string;
  confidence: "verified" | "partial" | "unverified";
  mode: "full" | "partial" | "placeholder";
  notes?: string | null;
  placeholder: {
    missing_fields: string[];
  };
};

function badgeClasses(confidence: PersonCard["confidence"]) {
  switch (confidence) {
    case "verified":
      return "border-emerald-700 bg-emerald-950 text-emerald-300";
    case "partial":
      return "border-amber-700 bg-amber-950 text-amber-300";
    case "unverified":
      return "border-zinc-700 bg-zinc-900 text-zinc-300";
    default:
      return "border-zinc-700 bg-zinc-900 text-zinc-300";
  }
}

export function PeopleRegistryCard({ person }: { person: PersonCard }) {
  return (
    <section className="rounded-2xl border border-zinc-800 bg-zinc-950 p-5 shadow-sm">
      <div className="flex items-start justify-between gap-3">
        <div>
          <h3 className="text-lg font-semibold text-zinc-100">{person.label}</h3>
          <p className="mt-1 text-sm text-zinc-400">
            {person.relationship || "other"} · {person.mode} profile
          </p>
        </div>

        <span
          className={`rounded-full border px-3 py-1 text-xs font-medium ${badgeClasses(
            person.confidence
          )}`}
        >
          {person.confidence}
        </span>
      </div>

      {person.notes ? (
        <p className="mt-4 text-sm leading-6 text-zinc-300">{person.notes}</p>
      ) : null}

      {person.placeholder.missing_fields.length ? (
        <div className="mt-4">
          <p className="text-xs uppercase tracking-[0.2em] text-zinc-500">
            Missing fields
          </p>
          <ul className="mt-2 space-y-2 text-sm text-zinc-300">
            {person.placeholder.missing_fields.map((field) => (
              <li key={field}>• {field}</li>
            ))}
          </ul>
        </div>
      ) : null}
    </section>
  );
}
